# P5 Agent（对话驱动画布）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把参考仓的 Eino Agent（对话、内置 Skill、子 Agent 媒体分析、画布能力清单、动态生成工具）搬进 sidecar，跑在 SQLite 上，通过 `/api/v1/agent/ws` 与前端 Agent 面板对话驱动画布；同一个 Hub 承载生成回写用的 `canvas_changed` 帧。

**Architecture:** `server/internal/agent/**` 从参考仓整体复制（A1：运行时不改），只做四类改动：① 仓库 SQL 改成 SQLite 方言；② 删多人协作（presence / 瞬态帧，X2）；③ 计费换空实现 `NoopBilling`；④ 加一个"发给上游大模型的媒体地址"接缝（S12 未定，图片先转 data URI，视频/音频给明确报错）。Skill、分类、工具、大模型四张表的内置数据由本计划的 `agentseed` 包 go:embed 并在每次启动幂等导入。HTTP 层移植 agent / skills / 后台大模型切换三组 handler，WS 不再验 JWT（localauth 已验 `?token=`），uid 恒为 1。

**Tech Stack:** Go 1.26（module `github.com/james/huabu_studio/apps/api`）、`github.com/cloudwego/eino` v0.8.13、`github.com/cloudwego/eino-ext/components/model/openai` v0.1.13、`github.com/coder/websocket` v1.8.14、`github.com/google/uuid` v1.6.0、`modernc.org/sqlite` v1.60.1；前端 React 18 + vitest。

**Spec:** `docs/specs/2026-09-29-canvas-desktop-spec.md`（F4、A1–A5、X2、S12、M4、M8）；事实依据 `docs/superpowers/plans/p1-research.md` 第 3、4、6 节。

```go
package agent

import (
	"sync"
	"time"
)

// agentMicroLayout 是 agent_messages.created_at / summarized_at 的库内格式（UTC，定宽 26 字符）。
// 定宽让 SQLite 的文本比较等价于时间比较，keyset 分页直接比库内原文。
const agentMicroLayout = "2006-01-02 15:04:05.000000"

var (
	microMu   sync.Mutex
	microLast time.Time
)

// agentMicroNow 返回严格单调递增的微秒时间文本：同一微秒内的第二次调用加 1µs，
// 保证同一进程里先写的消息排序一定在前（参考仓靠 MySQL NOW(6)，同刻时只能靠 id 碰运气）。
func agentMicroNow() string {
	microMu.Lock()
	defer microMu.Unlock()
	now := time.Now().UTC().Truncate(time.Microsecond)
	if !now.After(microLast) {
		now = microLast.Add(time.Microsecond)
	}
	microLast = now
	return now.Format(agentMicroLayout)
}
```

```go
			return err
```
```go
		}

func readJSONL[T any](name string) ([]T, error) {
	raw, err := dataFS.ReadFile("data/" + name + ".jsonl")
	if err != nil {
		return nil, err
	}
	var out []T
	sc := bufio.NewScanner(bytes.NewReader(raw))
	sc.Buffer(make([]byte, 0, 1<<20), 16<<20) // 单行 Skill 提示词可达数十 KB
	for sc.Scan() {
		line := bytes.TrimSpace(sc.Bytes())
		if len(line) == 0 {
			continue
		}
		var v T
		if err := json.Unmarshal(line, &v); err != nil {
			return nil, fmt.Errorf("%s.jsonl: %w", name, err)
		}
		out = append(out, v)
	}
	return out, sc.Err()
}

// jsonText 把种子里的 JSON 值转成库里的 TEXT；null / 缺失返回 nil（落 NULL）或 fallback。
func jsonText(v json.RawMessage, fallback string) any {
	s := strings.TrimSpace(string(v))
	if s == "" || s == "null" {
		if fallback == "" {
			return nil
		}
		return fallback
	}
	return s
}

func placeholders(n int) string { return strings.TrimSuffix(strings.Repeat("?,", n), ",") }

		return res, err

	if err != nil {
		return res, err
	}
	tools, err := readJSONL[toolRow]("agent_tools")
	if err != nil {
		return res, err
	}
	backbones, err := readJSONL[backboneRow]("agent_backbone_models")
	if err != nil {
		return res, err
	}

	if err != nil {
		return res, err

		return res, err
```

```go
package agent

import (
	"context"
	"encoding/base64"
	"fmt"
	"io"
	"net/http"
	"strings"
)

// MediaKeyExtractor 把本机媒体地址（sidecar 签出的 http URL 或 media://key）认回存储 key；
// 认不出（外链）返回 false。P1b 的 assets.Storage 直接满足。
type MediaKeyExtractor interface {
	ExtractKey(urlOrKey string) (string, bool)
}

const defaultMaxLLMImageBytes = 10 << 20

// LocalMediaResolver 是 S12 未定期间的实现：
//   - 外链 / data URI：原样放行（上游自己能拿到）；
//   - 本机图片：读出字节转 data URI（多数视觉模型接受）；
//   - 本机视频 / 音频：返回 ErrMediaNeedsPublicURL——上游只认公网地址，方案待定。
type LocalMediaResolver struct {
	Keys          MediaKeyExtractor
	Objects       AttachmentObjectReader
	MaxImageBytes int64 // 0 = 10MB
}

func (r *LocalMediaResolver) ResolveForLLM(ctx context.Context, kind, u string) (string, error) {
	if strings.HasPrefix(u, "data:") || r.Keys == nil {
		return u, nil
	}
	key, ok := r.Keys.ExtractKey(u)
	if !ok {
		return u, nil
	}
	if kind != "image" && kind != "" {
		return "", ErrMediaNeedsPublicURL
	}
	if r.Objects == nil {
		return "", fmt.Errorf("本机图片无法读取：存储未接入")
	}
	limit := r.MaxImageBytes
	if limit <= 0 {
		limit = defaultMaxLLMImageBytes
	}
	rc, err := r.Objects.GetObject(ctx, key)
	if err != nil {
		return "", fmt.Errorf("读取本机图片失败: %w", err)
	}
	defer rc.Close()
	data, err := io.ReadAll(io.LimitReader(rc, limit+1))
	if err != nil {
		return "", fmt.Errorf("读取本机图片失败: %w", err)
	}
	if int64(len(data)) > limit {
		return "", fmt.Errorf("图片超过 %d MB，无法发送给模型", limit>>20)
	}
	// 生成产物的 key 没有扩展名，按内容嗅探 MIME。
	mime := http.DetectContentType(data)
	if !strings.HasPrefix(mime, "image/") {
		return "", fmt.Errorf("文件不是图片（%s），无法作为图片发送给模型", mime)
	}
	return "data:" + mime + ";base64," + base64.StdEncoding.EncodeToString(data), nil
}

// resolveMediaForLLM 返回 msgs 的浅拷贝，其中 image_url 部件换成上游拿得到的地址。
// 不改原 msgs：它们还要参与后续轮次，且持久化必须保留原地址。解析失败的图片换成
// 一段文字说明，让 planner 知道这张图没送到，而不是整轮失败。
func (a *Agent) resolveMediaForLLM(ctx context.Context, msgs []*schema.Message) []*schema.Message {
	if a.MediaResolver == nil {
		return msgs
	}
	out := make([]*schema.Message, len(msgs))
	for i, m := range msgs {
		if m == nil || len(m.MultiContent) == 0 {
			out[i] = m
			continue
		}
		cp := *m
		parts := make([]schema.ChatMessagePart, 0, len(m.MultiContent))
		for _, p := range m.MultiContent {
			if p.Type == schema.ChatMessagePartTypeImageURL && p.ImageURL != nil {
				u, err := a.MediaResolver.ResolveForLLM(ctx, "image", p.ImageURL.URL)
				if err != nil {
					parts = append(parts, schema.ChatMessagePart{Type: schema.ChatMessagePartTypeText, Text: "[图片未能发送给模型：" + err.Error() + "]"})
					continue
				}
				img := *p.ImageURL
				img.URL = u
				p.ImageURL = &img
			}
			parts = append(parts, p)
		}
		cp.MultiContent = parts
		out[i] = &cp
	}
	return out
}
```

```go
package http

import (
	"errors"
	nethttp "net/http"

	"github.com/james/huabu_studio/apps/api/internal/agent"

// AdminAgentHandler 是设置页"Agent 模型"（spec A2、M8）的后端：大模型全部内置，
// 用户只能把某个角色的 model_name 换成另一个内置的已启用文本模型，或恢复默认。
type AdminAgentHandler struct {
	Repo     agent.BackboneRepo
	Switcher agent.BackboneModelSwitcher
	Models   agent.ModelLookup
	// Defaults 返回某行的内置 model_name（装配时传 agentseed.DefaultBackboneModelName）。
	Defaults func(id int64) (string, bool)
}

func (h *AdminAgentHandler) List(w nethttp.ResponseWriter, _ *nethttp.Request) {
	out, err := h.Repo.List()
	if err != nil {
		writeError(w, nethttp.StatusInternalServerError, err.Error())
		return
	}
	writeJSON(w, nethttp.StatusOK, map[string]any{"data": out})
}

// Update 只认 body 里的 model_name；其余字段（参考仓后台页会整行提交）一律忽略。
func (h *AdminAgentHandler) Update(w nethttp.ResponseWriter, r *nethttp.Request) {
	id, err := strconv.ParseInt(chi.URLParam(r, "id"), 10, 64)
	if err != nil {
		writeError(w, nethttp.StatusBadRequest, "bad id")
		return
	}
	var body struct {
		ModelName string `json:"model_name"`
	}
	if err := decodeJSON(r, &body); err != nil {
		writeError(w, nethttp.StatusBadRequest, err.Error())
		return
	}
	name := strings.TrimSpace(body.ModelName)
	if m, merr := h.Models.GetByModelID(name); name == "" || merr != nil || m == nil || m.Type != "text" || !m.Enabled {
		writeError(w, nethttp.StatusBadRequest, "只能切换到内置的已启用文本模型")
		return
	}
	h.applyAndRespond(w, id, h.Switcher.SetModelName(id, name))
}

func (h *AdminAgentHandler) Reset(w nethttp.ResponseWriter, r *nethttp.Request) {
	id, err := strconv.ParseInt(chi.URLParam(r, "id"), 10, 64)
	if err != nil {
		writeError(w, nethttp.StatusBadRequest, "bad id")
		return
	}
	def, ok := h.Defaults(id)
	if !ok {
		writeError(w, nethttp.StatusNotFound, "not found")
		return
	}
	h.applyAndRespond(w, id, h.Switcher.ResetModelName(id, def))
}

func (h *AdminAgentHandler) applyAndRespond(w nethttp.ResponseWriter, id int64, err error) {
	if errors.Is(err, agent.ErrBackboneNotFound) {
		writeError(w, nethttp.StatusNotFound, "not found")
		return
	}
	if err != nil {
		writeError(w, nethttp.StatusInternalServerError, err.Error())
		return
	}
	m, err := h.Repo.Get(id)
	if err != nil {
		writeError(w, nethttp.StatusInternalServerError, err.Error())
		return
	}
	writeJSON(w, nethttp.StatusOK, m)
}
```

```go
func (e e2e) sendAndCollect(t *testing.T, threadID, text string) []agent.Frame {
	t.Helper()
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	url := "ws" + strings.TrimPrefix(e.srv.URL, "http") + fmt.Sprintf("/api/v1/agent/ws?token=%s&project_id=%d", e2eToken, e.projectID)
	c, _, err := websocket.Dial(ctx, url, &websocket.DialOptions{HTTPHeader: nethttp.Header{"Origin": {"http://localhost:5173"}}})
	if err != nil {
		t.Fatal(err)
	}
	defer c.Close(websocket.StatusNormalClosure, "")
	buf, _ := json.Marshal(agent.Frame{Type: agent.FrameUserMessage, ThreadID: threadID, ActionID: "act1", Content: text})
	if err := c.Write(ctx, websocket.MessageText, buf); err != nil {
		t.Fatal(err)
	}
	var frames []agent.Frame
	for {
		_, data, err := c.Read(ctx)
		if err != nil {
			t.Fatalf("读帧失败（已收 %d 帧）: %v", len(frames), err)
		}
		var f agent.Frame
		_ = json.Unmarshal(data, &f)
		frames = append(frames, f)
		if f.Type == agent.FrameThreadEnd {
			return frames
		}
	}
}
```
